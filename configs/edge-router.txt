! ============================================
! EDGE ROUTER CONFIGURATION
! Cisco ISR - Internet Gateway
! ============================================
!
hostname EDGE-ROUTER-01
!
! --- WAN Interface ---
interface GigabitEthernet0/0/0
 description ISP Connection
 ip address dhcp
 ip nat outside
 no shutdown
!
! --- LAN Interface ---
interface GigabitEthernet0/0/1
 description LAN to Core Switch
 ip address 192.168.10.2 255.255.255.192
 ip nat inside
 no shutdown
!
! --- NAT / PAT Configuration ---
ip nat inside source list 100 interface GigabitEthernet0/0/0 overload
!
! --- ACL for NAT ---
access-list 100 permit ip 192.168.10.0 0.0.0.63 any
access-list 100 permit ip 192.168.20.0 0.0.0.255 any
access-list 100 permit ip 192.168.30.0 0.0.1.255 any
access-list 100 permit ip 192.168.50.0 0.0.0.255 any
! NOTE: Guest VLAN 40 intentionally excluded from NAT - captive portal required
!
! --- Guest Isolation ACL ---
access-list 110 deny ip 192.168.40.0 0.0.0.255 192.168.10.0 0.0.0.63
access-list 110 deny ip 192.168.40.0 0.0.0.255 192.168.20.0 0.0.0.255
access-list 110 deny ip 192.168.40.0 0.0.0.255 192.168.30.0 0.0.1.255
access-list 110 deny ip 192.168.40.0 0.0.0.255 192.168.50.0 0.0.0.255
access-list 110 permit ip 192.168.40.0 0.0.0.255 any
!
interface GigabitEthernet0/0/1
 ip access-group 110 in
!
! --- Default Route via DHCP ---
ip route 0.0.0.0 0.0.0.0 GigabitEthernet0/0/0 dhcp
!
! --- SSH Hardening ---
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
!
end
